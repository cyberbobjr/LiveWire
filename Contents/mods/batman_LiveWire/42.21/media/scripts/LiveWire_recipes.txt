       accepts calculators but only gives electronics scrap. */

    {
        timedAction = DismantleElectrical,
        Icon = Item_Calculator,
        time = 60,
        Tags = InHandCraft;Electrical,
        category = Electrical,
        xpAward = Electricity:3,
        inputs
        {
            item 1 tags[base:screwdriver] mode:keep flags[NoBrokenItems],
            item 1 [Base.Calculator] mode:destroy,
        }
        outputs
        {
            item 1 Base.ElectronicsScrap,
            item 1 LiveWire.SolarCell,
        }
    }
